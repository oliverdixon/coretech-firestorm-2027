\begin{frame}
    \frametitle{Implementation Strategy}
    \begin{itemize}
        \item Monolithic C++23 POSIX application
        \item Multi-threaded network transport in the number of sinks
        \item Worker threads accept tasks in a bounded queue\pause
        \item Extensive use of STL
            \begin{itemize}
                \item Cache-friendly data storage
                \item No-copy functional-style constructs with lazy operations
                \item Explicit ownership and RAII manage heap-allocated data
                \item Static storage allocation and compile-time evaluation
            \end{itemize}\pause
        \item Failure modes enforced with exceptions and exception guarantees
    \end{itemize}
\end{frame}
\begin{frame}
    \frametitle{Validation Strategy}
    \begin{itemize}[<+->]
        \item Unit tests across fixtures to exercise each component:
            \begin{itemize}[<.->]
                \item Underlying connection
                \item Frame routers
                \item Frame sinks
                \item Traffic decoder
                \item Semantics of data model
            \end{itemize}
        \item End-to-end tests provided by CoreTech.
            \begin{itemize}[<.->]
                \item Exercises the happy path
                \item Constructed documents verified by SHA256
            \end{itemize}
        \item Integrated into CI pipeline to identify correctness regressions
        \item Static analysis (LLVM tools) and dynamic analysis (Valgrind Memcheck)
        \item Extensive documentation (Doxygen) to aid review
    \end{itemize}
\end{frame}
